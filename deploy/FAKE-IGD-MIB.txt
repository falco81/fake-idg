-- fake-igd, the fake UPnP router (IGD) that opens no port: what its own SNMPv3
-- agent answers (1.7.2) - the pages' API answers it (Settings, Monitoring). Read
-- only. Under netSnmpPlaypen (net-snmp's subtree for objects of one's own), .8 -
-- fake-igd has no enterprise number of its own (otp, its model, is .7).
-- Every sample of https://SERVER/metrics is in fakeIgdMetricTable too.

IMPORTS
    MODULE-IDENTITY, OBJECT-TYPE, NOTIFICATION-TYPE,
    Integer32, Gauge32, Counter64, TimeTicks
        FROM SNMPv2-SMI
    DisplayString
        FROM SNMPv2-TC
    OBJECT-GROUP, NOTIFICATION-GROUP, MODULE-COMPLIANCE
        FROM SNMPv2-CONF
    netSnmpPlaypen
        FROM NET-SNMP-MIB;

fakeIgd MODULE-IDENTITY
    LAST-UPDATED "202611160000Z"
    ORGANIZATION "fake-igd"
    CONTACT-INFO "The administrator of the fake-igd server - Settings, Monitoring."
    DESCRIPTION  "fake-igd: the services, the requests and the ports asked for, the
                  devices, the mappings, the mDNS reflector, the alerts, the
                  certificate, the machine, the mail and the logs of one server."
    REVISION     "202611160000Z"
    DESCRIPTION  "First: 1.7.2."
    ::= { netSnmpPlaypen 8 }
